module uart_tx(
  input PCLK, PRESETn,
  input [7:0] PWDATA,
  input tx_fifo_push, enable,
  input [7:0] LCR,
  output wire [4:0] tx_fifo_count,
  output reg busy, 
  output wire tx_fifo_empty, tx_fifo_full,
  output wire TXD
);
  
  // TX FSM
  localparam IDLE   = 4'b0000, //0
  			START  = 4'b0001, //1
  			BIT0   = 4'b0010, //2
  			BIT1   = 4'b0011, //3
  			BIT2   = 4'b0100, //4
  			BIT3   = 4'b0101, //5
  			BIT4   = 4'b0110, //6
  			BIT5   = 4'b0111, //7
  			BIT6   = 4'b1000, //8
  			BIT7   = 4'b1001, //9
  			PARITY = 4'b1010, //10
  			STOP1  = 4'b1011, //11
  			STOP2  = 4'b1100; //12
  
  reg [3:0] tx_state, bit_counter;
  reg [7:0] tx_buffer;
  wire [7:0] tx_fifo_out;
  reg pop_tx_fifo, TXD_tmp;
  
  reg parity;
  
  uart_fifo tx_fifo(.data_in(PWDATA), .clk(PCLK), .rstn(PRESETn), .push(tx_fifo_push), .pop(pop_tx_fifo), .data_out(tx_fifo_out), .fifo_empty(tx_fifo_empty), .fifo_full(tx_fifo_full), .count(tx_fifo_count));
  
  //baud pulse
  always@(posedge PCLK)
    begin
      if(!PRESETn)
        bit_counter <= 4'b0000;
      else if(enable)
        bit_counter <= bit_counter + 4'b0001;
    end
  
  //internal fifo input 'pop'
  always@(posedge PCLK)
    begin
      if(!PRESETn) 
        pop_tx_fifo <= 1'b0;
      else if(!tx_fifo_empty & tx_state == START & (bit_counter == 4'h8) & enable)
      	pop_tx_fifo <= 1'b1;
      else
        pop_tx_fifo <= 1'b0;
    end
  
  //FSM
  always@(posedge PCLK)
    begin
      if(!PRESETn)
        tx_state <= IDLE;
      else if(enable)
        case(tx_state)
          IDLE :if((bit_counter == 4'hf) & enable & (tx_fifo_empty == 1'b0)) begin tx_state <= START; busy <= 1'b1; end // pop_tx_fifo <= 1'b1; end

          START :if((bit_counter == 4'hf) & enable) tx_state <= BIT0;

          BIT0 :if((bit_counter == 4'hf) & enable) tx_state <= BIT1;
//           else if(enable == 1'b1) tx_state <= BIT0;

          BIT1 :if((bit_counter == 4'hf) & enable) tx_state <= BIT2;
//           else if(enable == 1'b1) tx_state <= BIT1;

          BIT2 :if((bit_counter == 4'hf) & enable) tx_state <= BIT3;
//           else if(enable == 1'b1) tx_state <= BIT2;

          BIT3 :if((bit_counter == 4'hf) & enable) tx_state <= BIT4;
//           else if(enable == 1'b1) tx_state <= BIT3;

          BIT4 :if((bit_counter == 4'hf) & enable & (LCR[1:0] > 2'b00)) tx_state <= BIT5;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b00) & (LCR[3] == 1'b1)) tx_state <= PARITY;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b00) & (LCR[3] == 1'b0))  tx_state <= STOP1;
//           else if(enable == 1'b1) tx_state <= BIT4;

          BIT5 :if((bit_counter == 4'hf) & enable & (LCR[1:0] > 2'b01)) tx_state <= BIT6;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b01) & (LCR[3] == 1'b1)) tx_state <= PARITY;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b01) & (LCR[3] == 1'b0)) tx_state <= STOP1;
//           else if(enable == 1'b1) tx_state <= BIT5;

          BIT6 :if((bit_counter == 4'hf) & enable & (LCR[1:0] > 2'b10)) tx_state <= BIT7;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b10) & (LCR[3] == 1'b1)) tx_state <= PARITY;
          else if((bit_counter == 4'hf) & enable & (LCR[1:0] == 2'b10) & (LCR[3] == 1'b0)) tx_state <= STOP1;

          BIT7 :if((bit_counter == 4'hf) & enable & (LCR[3] == 1'b1)) tx_state <= PARITY;
          else if((bit_counter == 4'hf) & enable & (LCR[3] == 1'b0)) tx_state <= STOP1;

          PARITY :if((bit_counter == 4'hf) & enable) tx_state <= STOP1;
          else if((bit_counter == 4'hf) & enable & (LCR[2] == 1'b1)) tx_state <= STOP2; 

          STOP1 :if((bit_counter == 4'hf) & enable & (LCR[2] == 1'b1)) tx_state <= STOP2;
          else if((bit_counter == 4'hf) & enable & (LCR[2] == 1'b0)) tx_state <= IDLE; 

          STOP2 : if((bit_counter == 4'hf) & enable) tx_state <= IDLE;

          default: tx_state <= IDLE;
        endcase
    end
      
  //TXD_tmp
  always@(tx_state)
    begin
      if(tx_state == IDLE)
        TXD_tmp <= 1'b1;
      else if(tx_state == START)
        TXD_tmp <= 1'b0;
      else if(tx_state == BIT0 || tx_state == BIT1 || tx_state == BIT2 || tx_state == BIT3 || tx_state == BIT4 || tx_state == BIT5 || tx_state == BIT6 || tx_state == BIT7)
        TXD_tmp <= tx_buffer[0];
      else if(tx_state == PARITY)
        TXD_tmp <= parity;
      else if(tx_state == STOP1 || tx_state == STOP2)
        TXD_tmp <= 1'b1;
    end
  
  //busy
  always@(posedge PCLK)
    begin
      if(!PRESETn)
        busy <= 1'b0;
      else if(tx_state == IDLE)
        busy <= 1'b0;
      else
        busy <= 1'b1;
    end
  
  //tx_buffer
  always@(negedge pop_tx_fifo)
    begin
      if(tx_state == START)
        begin
          case({LCR[1:0]})
            2'b00 : tx_buffer[4:0] <= tx_fifo_out[4:0];
            2'b01 : tx_buffer[5:0] <= tx_fifo_out[5:0];
            2'b10 : tx_buffer[6:0] <= tx_fifo_out[6:0];
            2'b11 : tx_buffer[7:0] <= tx_fifo_out[7:0];
            default tx_buffer[7:0] <= tx_fifo_out[7:0];
          endcase
        end
    end
  
  //tx_buffer
  always@(tx_state)
    begin
      if(tx_state == BIT0 || tx_state == BIT1 || tx_state == BIT2 || tx_state == BIT3 || tx_state == BIT4 || tx_state == BIT5 || tx_state == BIT6 || tx_state == BIT7 & (bit_counter == 4'hf))
        tx_buffer <= tx_buffer >> 1;
    end
  
  //parity
  always@(tx_buffer)
    begin
      if(tx_state == START)
      begin
        case({LCR[5:4]})
          2'b00 : parity <= ~(^tx_buffer); //odd parity generator
          2'b01 : parity <=  (^tx_buffer); //even parity generator 
          2'b10 : parity <= 1'b1; //odd parity but LCR[4] inverse
          2'b11 : parity <= 1'b0; //even parity but LCR[4] inverse
          default  parity <= 1'b0;
        endcase
      end
    end
  
  //TXD & LCR bit-6 checking
  assign TXD = LCR[6] ? 1'b0: TXD_tmp;
endmodule
